# MiniLang SSA IR Program

function factorial: int (n)

entry:
  t0 = CMP_LE n00, 1
  JUMP_IF t0, then_1
  JUMP else_2

then_1:
  RETURN 1

else_2:
  JUMP endif_3

endif_3:
  t4 = SUB n00, 1
  PARAM t4
  t5 = CALL factorial
  t6 = MUL n00, t5
  RETURN t6

function main: int ()

entry:
  PARAM 5
  t0 = CALL factorial
  RETURN t0
